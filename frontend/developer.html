predict_sentiment.py 
import requests

API_URL = "http://localhost:8000/predict/custom"
API_KEY = "sk_live_YOUR_KEY_HERE"  # ← paste your key

def analyse_sheng(text: str) -> dict:
    response = requests.post(
        API_URL,
        headers={"X-API-Key": API_KEY},
        json={"text": text},
        timeout=10
    )
    response.raise_for_status()
    return response.json()

# Example usage
result = analyse_sheng("Mpesa ilishindwa tena! Shida kubwa!")

print(f"Label: {result['label']}")
print(f"Score: {result['score']:.2%}")
print(f"Keywords: {', '.join(result['keywords'])}")
print(f"Latency: {result['processing_time_ms']}ms")
shengsense.js 
const API_URL = 'http://localhost:8000/predict/custom';
const API_KEY = 'sk_live_YOUR_KEY_HERE';  // ← paste your key

async function analyseSheng(text) {
  const res = await fetch(API_URL, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'X-API-Key': API_KEY
    },
    body: JSON.stringify({ text })
  });

  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return res.json();
}

// Example usage
analyseSheng('Poa sana! Safaricom ni top kabisa.')
  .then(data => {
    console.log('Label:', data.label);
    console.log('Score:', data.score);
    console.log('Keywords:', data.keywords);
    console.log('Latency (ms):', data.processing_time_ms);
  })
  .catch(console.error);

Method Endpoint Auth Description

Response Schema

{
  "label":              string,   // "POSITIVE" | "NEGATIVE" | "NEUTRAL"
  "score":              float,    // 0.0 – 1.0 confidence
  "model":              string,   // model identifier
  "processing_time_ms": float,    // inference latency
  "keywords":           string[]  // Sheng/Swahili tokens extracted
}
